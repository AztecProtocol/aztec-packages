# bb-avm-sim: the AVM simulator as a service over avm_schema.json. Two artifacts carry it to
# @aztec-foundation/bb-avm-sim (barretenberg/ts/bb-avm-sim):
#   - the bb-avm-sim binary (native builds), served over IPC, reaching world state and contract
#     data through its wsdb and cdb IPC clients;
#   - bb-avm-sim.wasm, a WASI reactor (wasm builds) exporting the generated FFI entry
#     (avm_ipc_ffi_entry). It has no route to world state or contract data yet, so it serves
#     SimulateWithHints alone.
set(AVM_SCHEMA ${CMAKE_CURRENT_SOURCE_DIR}/avm_schema.json)
set(AVM_GEN_DIR ${CMAKE_CURRENT_SOURCE_DIR}/generated)
set(AVM_GEN_OUTPUTS
    ${AVM_GEN_DIR}/avm_dispatch.hpp
    ${AVM_GEN_DIR}/avm_ffi.cpp
    ${AVM_GEN_DIR}/avm_ffi.hpp
    ${AVM_GEN_DIR}/avm_ipc_server.hpp
    ${AVM_GEN_DIR}/avm_types.hpp
    ${AVM_GEN_DIR}/ipc_codegen/msgpack_adaptor.hpp
    ${AVM_GEN_DIR}/ipc_codegen/msgpack_include.hpp
    ${AVM_GEN_DIR}/ipc_codegen/throw.hpp
)
set(IPC_CODEGEN_DIR ${CMAKE_SOURCE_DIR}/../../ipc-codegen)
file(GLOB_RECURSE IPC_CODEGEN_SRC
    ${IPC_CODEGEN_DIR}/src/*.ts
    ${IPC_CODEGEN_DIR}/templates/cpp/*.hpp
)
add_custom_command(
    OUTPUT ${AVM_GEN_OUTPUTS}
    COMMAND node --experimental-strip-types --no-warnings
            ${IPC_CODEGEN_DIR}/src/generate.ts
            --schema ${AVM_SCHEMA}
            --lang cpp
            --out ${AVM_GEN_DIR}
            --server
            --ffi
            --cpp-namespace bb::avm
            --cpp-include-dir barretenberg/avm/generated
            --prefix Avm
            --strip-method-prefix
    DEPENDS ${AVM_SCHEMA} ${IPC_CODEGEN_SRC}
    COMMENT "Generating the AVM IPC server and FFI entry from avm_schema.json"
    VERBATIM
)
add_custom_target(avm_ipc_generated DEPENDS ${AVM_GEN_OUTPUTS})

# The tests read vm2's minimal transaction fixture, embedded so the wasm build needs no filesystem.
set(MINIMAL_TX_FIXTURE ${CMAKE_CURRENT_SOURCE_DIR}/../vm2/testing/minimal_tx.testdata.bin)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${MINIMAL_TX_FIXTURE})
file(READ ${MINIMAL_TX_FIXTURE} MINIMAL_TX_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," MINIMAL_TX_BYTES "${MINIMAL_TX_HEX}")
set(AVM_TEST_INCLUDE_DIR ${CMAKE_CURRENT_BINARY_DIR}/include)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/minimal_tx_fixture.hpp.tmp
    "#pragma once\n// Generated by barretenberg/avm/CMakeLists.txt from vm2/testing/minimal_tx.testdata.bin.\n"
    "inline constexpr unsigned char MINIMAL_TX_FIXTURE[] = { ${MINIMAL_TX_BYTES} };\n")
# Copied only when it changes, so a reconfigure does not rebuild the tests.
configure_file(${CMAKE_CURRENT_BINARY_DIR}/minimal_tx_fixture.hpp.tmp
    ${AVM_TEST_INCLUDE_DIR}/barretenberg/avm/minimal_tx_fixture.hpp COPYONLY)

# The service as a library over the generated FFI entry, which the wasm reactor exports and the
# tests call. The entry is generated at build time, so a clean tree's configure cannot glob it.
barretenberg_module_with_sources(
    avm_ffi
    SOURCE_FILES
        ${CMAKE_CURRENT_SOURCE_DIR}/avm_ffi_dispatcher.cpp
        ${CMAKE_CURRENT_SOURCE_DIR}/avm_simulate_with_hints.cpp
        ${AVM_GEN_DIR}/avm_ffi.cpp
    HEADER_FILES ${CMAKE_CURRENT_SOURCE_DIR}/avm_simulate_with_hints.hpp
    TEST_SOURCE_FILES ${CMAKE_CURRENT_SOURCE_DIR}/avm_ffi.test.cpp
    DEPENDENCIES vm2_sim
)
add_dependencies(avm_ffi_objects avm_ipc_generated)
if(TARGET avm_ffi_test_objects)
    add_dependencies(avm_ffi_test_objects avm_ipc_generated)
    target_include_directories(avm_ffi_test_objects PRIVATE ${AVM_TEST_INCLUDE_DIR})
endif()

if(WASM)
    # A WASI reactor exporting the FFI entry: the module and its dependencies plus barretenberg/wasi
    # (the env functions over WASI, and _initialize). It asks its host for WASI alone.
    add_executable(
        bb-avm-sim-debug.wasm
        $<TARGET_OBJECTS:avm_ffi_objects>
        $<TARGET_OBJECTS:wasi_objects>
    )
    target_link_libraries(bb-avm-sim-debug.wasm PRIVATE vm2_sim)
    target_link_options(bb-avm-sim-debug.wasm PRIVATE -nostartfiles -Wl,--no-entry,--export-dynamic)
    add_custom_command(
        OUTPUT ${CMAKE_BINARY_DIR}/bin/bb-avm-sim.wasm
        COMMAND ${CMAKE_STRIP} ${CMAKE_BINARY_DIR}/bin/bb-avm-sim-debug.wasm -o ${CMAKE_BINARY_DIR}/bin/bb-avm-sim.wasm
        DEPENDS bb-avm-sim-debug.wasm
        COMMENT "Stripping debug info to create bb-avm-sim.wasm"
    )
    add_custom_target(bb-avm-sim.wasm ALL DEPENDS ${CMAKE_BINARY_DIR}/bin/bb-avm-sim.wasm)
    return()
endif()

add_executable(
    bb-avm-sim
    main.cpp
    cli.cpp
    avm_execute.cpp
    avm_ipc_server.cpp
)
add_dependencies(bb-avm-sim avm_ipc_generated)
target_include_directories(
    bb-avm-sim
    PRIVATE
    ${AVM_GEN_DIR}
)
target_link_libraries(
    bb-avm-sim
    PRIVATE
    barretenberg
    avm_ffi
    vm2_sim
    wsdb_ipc_merkle_db
    cdb_ipc_client
    ipc_runtime
    env
)
if(ENABLE_STACKTRACES)
    target_link_libraries(
        bb-avm-sim
        PUBLIC
        Backward::Interface
    )
    target_link_options(
        bb-avm-sim
        PRIVATE
        -ldw -lelf
    )
endif()
